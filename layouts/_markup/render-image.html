{{- /* Markdown images, i.e. the sponsor logos: ![alt](/images/name.png) The file has to exist under assets/ (the build fails otherwise). The hook adds the pixel dimensions and scales anything wider than 800px down at build time, so a huge original can be committed as it came. No lazy loading: the logos sit near the top of the page. */ -}} {{- $path := strings.TrimPrefix "/" .Destination -}} {{- $image := resources.Get $path -}} {{- if not $image -}} {{- errorf "%s: image %q not found; expected the file assets/%s" .Page.File.Path .Destination $path -}} {{- else -}} {{- $raster := and (eq $image.ResourceType "image") (ne $image.MediaType.SubType "svg") -}} {{- if and $raster (gt $image.Width 800) -}} {{- $small := $image.Resize "800x" -}} {{- if lt (len $small.Content) (len $image.Content) -}} {{- $image = $small -}} {{- end -}} {{- end -}} {{- end -}}
